Key Design Decisions
Distinct curriculum entities instead of a polymorphic type
Rather than having a single CurriculumItem with a type enum, the platform uses seven separate entities — Subject, Course, Topic, Module, Quest, Checkpoint, and Raid — each with its own fields, relationships, and extension points. This avoids nullable field sprawl and allows each entity to evolve independently as the curriculum design matures.
Content as HTML
Module content, quest learning materials, and task prompts are stored as rich HTML, produced by a backend TinyMCE editor. The frontend simply renders whatever HTML it receives, without needing specialized components for videos, code blocks, images, or other media types. This keeps frontend complexity minimal and gives content authors maximum flexibility.
XP as an immutable ledger
Rather than a mutable xp_total field on the enrollment record, every XP change is recorded as an XPTransaction row. The current balance is derived. This provides a full audit trail and makes it trivial to explain where any student's points came from.
Gitea as the Git backend
The platform delegates all Git operations to a self-hosted Gitea instance rather than building its own Git infrastructure. The platform maintains thin mapping tables (GiteaAccount, GiteaRepository) to link platform concepts to Gitea entities, and listens to Gitea's push webhooks to trigger automated test runs.
Cascading configuration
All configurable behaviors — group sizes, XP cooldowns, notification content, submission rules — are stored in a ConfigurationSetting table with a scope hierarchy: Organization → Campus → Program → Cohort. Any level can override the one above it. This means a specific cohort can have different cooldown rules without touching the program defaults.
Separation of Record and RecordEffect
An admin Record (e.g., "block this student") documents the decision and reason. A RecordEffect row documents the system action taken (e.g., revoke platform access). The same record type can produce different effects in different programs, and effects can be toggled or extended without changing the record itself.

const ALL_PAGE_REFS = Object.keys(PAGES).map(id => ({
  id,
  title: PAGES[id].title,
  subtitle: PAGES[id].subtitle || ''
}));
function navigate(pageId) {
  window.location.hash = pageId;
}
function renderPage(pageId) {
  const page = PAGES[pageId] || PAGES['home'];
  // breadcrumb
  const bc = document.getElementById('breadcrumb');
  let breadHTML = 'Main page';
  if (page.breadcrumb && page.breadcrumb.length) {
    page.breadcrumb.forEach(crumb => {
      const crumbPage = PAGES[crumb];
      if (crumbPage) breadHTML += ` › ${crumbPage.title.split('—')[0].trim()}`;
    });
  }
  if (pageId !== 'home') breadHTML += ` › ${page.title}`;
  bc.innerHTML = breadHTML;
  document.getElementById('page-title').textContent = page.title;
  document.getElementById('page-subtitle').textContent = page.subtitle || '';
  document.getElementById('page-lead').innerHTML = page.lead || '';
  // update nav active state
  document.querySelectorAll('.nav-link').forEach(el => el.classList.remove('active'));
  // build body
  const body = document.getElementById('page-body');
  let html = '';
  // infobox
  if (page.infobox) {
    html += `
      ${page.infobox.title}`;
    if (page.infobox.subtitle) html += `${page.infobox.subtitle}`;
    html += ``;
    page.infobox.rows.forEach(row => {
      html += `${row.label}${row.value}`;
    });
    html += ``;
  }
  // TOC
  if (page.toc && page.toc.length > 0) {
    html += `Contents`;
    page.toc.forEach((item, i) => {
      html += `${item.title}`;
    });
    html += ``;
  }
  html += page.body || '';
  // related pages
  if (page.related && page.related.length) {
    html += `See also`;
    page.related.forEach(relId => {
      const relPage = PAGES[relId];
      if (relPage) {
        html += `${relPage.title.split('—')[0].trim()}`;
      }
    });
    html += ``;
  }
  body.innerHTML = html;
  // make all internal #hash links work through router
  body.querySelectorAll('a[href^="#"]').forEach(link => {
    const target = link.getAttribute('href').slice(1);
    if (PAGES[target]) {
      link.removeAttribute('href');
      link.style.cursor = 'pointer';
      link.onclick = () => navigate(target);
    }
  });
  window.scrollTo(0, 0);
  document.title = `${page.title} — TalentNation Platform Wiki`;
}

const searchInput = document.getElementById('search-input');
const searchResults = document.getElementById('search-results');

  const q = searchInput.value.trim().toLowerCase();

  const matches = ALL_PAGE_REFS.filter(p =>
    p.title.toLowerCase().includes(q) || p.subtitle.toLowerCase().includes(q)
  ).slice(0, 8);
  if (!matches.length) { searchResults.style.display = 'none'; return; }
  searchResults.innerHTML = matches.map(p =>
    `
      ${p.title}
      ${p.subtitle}
    `
  ).join('');
  searchResults.style.display = 'block';
});
document.getElementById('search-btn').onclick = () => {
  const q = searchInput.value.trim().toLowerCase();
  const match = ALL_PAGE_REFS.find(p => p.title.toLowerCase().includes(q));
  if (match) { navigate(match.id); searchInput.value = ''; searchResults.style.display = 'none'; }
};

  if (e.key === 'Enter') document.getElementById('search-btn').click();
  if (e.key === 'Escape') { searchResults.style.display = 'none'; }
